table alert_status_history (tabela lub widok: alert_status_history)
CREATE TABLE alert_status_history (
    history_id INTEGER PRIMARY KEY,
    alert_id   INTEGER NOT NULL REFERENCES alerts(alert_id) ON DELETE RESTRICT,
    old_status TEXT NOT NULL,
    new_status TEXT NOT NULL,
    changed_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
)
table alerts (tabela lub widok: alerts)
CREATE TABLE alerts (
    alert_id   INTEGER PRIMARY KEY,
    source_ip  TEXT NOT NULL,
    rule_name  TEXT NOT NULL,
    status     TEXT NOT NULL CHECK(status IN ('OPEN','INVESTIGATING','CLOSED','FALSE_POSITIVE')),
    first_seen TEXT NOT NULL,
    last_seen  TEXT NOT NULL,
    event_count INTEGER NOT NULL CHECK(event_count > 0),
    UNIQUE(source_ip, rule_name, first_seen)
)
table event_statistics (tabela lub widok: event_statistics)
CREATE TABLE event_statistics (
    source_ip   TEXT PRIMARY KEY,
    event_count INTEGER NOT NULL CHECK(event_count >= 0),
    last_seen   TEXT NOT NULL
)
table network_events (tabela lub widok: network_events)
CREATE TABLE network_events (
    event_id    INTEGER PRIMARY KEY,
    event_key   TEXT NOT NULL UNIQUE,
    timestamp   TEXT NOT NULL,
    sensor_id   INTEGER NOT NULL REFERENCES sensors(sensor_id) ON DELETE RESTRICT,
    src_ip      TEXT NOT NULL,
    dst_ip      TEXT NOT NULL,
    src_port    INTEGER CHECK(src_port BETWEEN 1 AND 65535),
    dst_port    INTEGER NOT NULL CHECK(dst_port BETWEEN 1 AND 65535),
    protocol    TEXT NOT NULL CHECK(protocol IN ('TCP', 'UDP', 'ICMP')),
    event_type  TEXT NOT NULL,
    action      TEXT NOT NULL CHECK(action IN ('ALLOW', 'BLOCK')),
    bytes       INTEGER NOT NULL DEFAULT 0 CHECK(bytes >= 0),
    duration_ms INTEGER NOT NULL DEFAULT 0 CHECK(duration_ms >= 0)
)
table sensors (tabela lub widok: sensors)
CREATE TABLE sensors (
    sensor_id   INTEGER PRIMARY KEY,
    sensor_name TEXT NOT NULL UNIQUE,
    location    TEXT NOT NULL DEFAULT 'unknown'
)
trigger log_alert_status_change (tabela lub widok: alerts)
CREATE TRIGGER log_alert_status_change
AFTER UPDATE OF status ON alerts
FOR EACH ROW
WHEN OLD.status <> NEW.status
BEGIN
    INSERT INTO alert_status_history (alert_id, old_status, new_status)
    VALUES (NEW.alert_id, OLD.status, NEW.status);
END
view suspicious_sources (tabela lub widok: suspicious_sources)
CREATE VIEW suspicious_sources AS
SELECT src_ip, COUNT(*) AS failed_logins, MIN(timestamp) AS first_failure,
       MAX(timestamp) AS last_failure
FROM network_events
WHERE event_type='AUTH_FAILURE'
GROUP BY src_ip HAVING COUNT(*) >= 5
